BTC 25Δ call · vol + Greeks 
from cryptovol import CryptoVol

cv = CryptoVol(api_key="cvk_live_...")

pt = cv.vol_surface(
    ccy="BTC",
    expiry="2026-09-26",
    strike_type="delta",
    strike_value=0.25,
    option_type="C",
    include_analytics=True,
)

g = pt.analytics.greeks
print(pt.vol, g.delta, g.gamma, g.vega)
